import 'dart:developer';

import 'package:dio/dio.dart';
import 'package:injectable/injectable.dart';
import 'package:shopping_app/core/storage/token_storage.dart';

@lazySingleton
class AuthInterceptor extends Interceptor {
  final TokenStorage _tokenStorage;

  AuthInterceptor(this._tokenStorage);

  @override
  Future<void> onRequest(
    RequestOptions options,
    RequestInterceptorHandler handler,
  ) async {
    try {
      final token = await _tokenStorage.getToken();

      final hasToken = token != null && token.trim().isNotEmpty;

      log('TOKEN EXISTS: $hasToken');
      log('TOKEN LENGTH: ${token?.trim().length ?? 0}');

      if (hasToken) {
        options.headers['Authorization'] = 'Bearer ${token!.trim()}';
      } else {
        options.headers.remove('Authorization');
      }

      handler.next(options);
    } catch (error, stackTrace) {
      log(
        'Failed to retrieve authentication token',
        error: error,
        stackTrace: stackTrace,
      );

      handler.reject(
        DioException(
          requestOptions: options,
          error: error,
        ),
      );
    }
  }
}

Future<String> initialRoute() async {
  final prefs = await SharedPreferences.getInstance();
  final bool seenOnboarding = prefs.getBool('seenOnboarding') ?? false;
  if (!seenOnboarding) {
    return RoutesManager.onBoarding;
  }
  final storage = serviceLocator<AppSecureStorage>();
  final token = await storage.read(AppKeys.token);

  if (token.isNotEmpty) {
    return RoutesManager.appSection;
  } else {
    return RoutesManager.hello;
  }
}

import 'package:shopping_app/features/cart/domain/entities/cart_item_entity.dart';

class CartDto {
  final int productId;
  final String title;
  final String image;
  final double price;
  final int quantity;

  const CartDto({
    required this.productId,
    required this.title,
    required this.image,
    required this.price,
    required this.quantity,
  });

  factory CartDto.fromJson(Map<String, dynamic> json) {
    final productValue = json['product'];

    final Map<String, dynamic> product = productValue is Map
        ? Map<String, dynamic>.from(productValue)
        : json;

    return CartDto(
      productId: _toInt(
        json['productId'] ?? json['product_id'] ?? product['id'] ?? json['id'],
      ),
      title: _toString(
        product['title'] ?? product['name'] ?? json['title'] ?? json['name'],
        fallback: 'Product',
      ),
      image: _toImage(
        product['image'] ??
            product['images'] ??
            product['thumbnail'] ??
            json['image'] ??
            json['images'],
      ),
      price: _toDouble(product['price'] ?? json['price']),
      quantity: _toInt(json['quantity'] ?? json['count'], fallback: 1),
    );
  }

  CartItemEntity toEntity() {
    return CartItemEntity(
      productId: productId,
      title: title,
      image: image,
      price: price,
      quantity: quantity < 1 ? 1 : quantity,
    );
  }

  static int _toInt(dynamic value, {int fallback = 0}) {
    if (value is int) return value;
    if (value is num) return value.toInt();

    return int.tryParse(value?.toString() ?? '') ?? fallback;
  }

  static double _toDouble(dynamic value) {
    if (value is num) return value.toDouble();

    return double.tryParse(value?.toString() ?? '') ?? 0.0;
  }

  static String _toString(dynamic value, {String fallback = ''}) {
    if (value == null) return fallback;

    final result = value.toString().trim();

    return result.isEmpty ? fallback : result;
  }

  static String _toImage(dynamic value) {
    if (value is String) return value;

    if (value is List && value.isNotEmpty) {
      return value.first?.toString() ?? '';
    }

    return '';
  }
}

class CartDataSourceImp implements CartDataSourceInterface {
  final CartApi _api;

  CartDataSourceImp(this._api);

  @override
  Future<ResultApi<List<CartItemEntity>>> getCart() async {
    try {
      final json = await _api.getCart();

      final rawList = json['list'];

      if (rawList != null && rawList is! List) {
        return Error<List<CartItemEntity>>('Invalid cart items format');
      }

      final items = rawList as List<dynamic>? ?? [];

      final cartItems = items.map((item) {
        if (item is! Map) {
          throw const FormatException('Invalid cart item format');
        }

        return CartDto.fromJson(Map<String, dynamic>.from(item)).toEntity();
      }).toList();

      return Success<List<CartItemEntity>>(cartItems);
    } catch (e) {
      return Error<List<CartItemEntity>>(_errorMessage(e));
    }
  }

  @override
  Future<ResultApi<void>> addToCart({required int productId}) async {
    try {
      await _api.addToCart(productId: productId);

      return Success<void>(null);
    } catch (e) {
      return Error<void>(_errorMessage(e));
    }
  }

  @override
  Future<ResultApi<void>> deleteCartItem({required int productId}) async {
    try {
      await _api.deleteCartItem(productId: productId);

      return Success<void>(null);
    } catch (e) {
      return Error<void>(_errorMessage(e));
    }
  }

  String _errorMessage(Object error) {
    if (error is DioException) {
      final responseData = error.response?.data;

      if (responseData is Map) {
        final message = responseData['message'];

        if (message != null) {
          return message.toString();
        }
      }

      if (error.response?.statusCode != null) {
        return 'Request failed with status '
            '${error.response!.statusCode}';
      }

      return error.message ?? 'Network error';
    }

    return error.toString();
  }

@injectable
class CartCubit extends Cubit<CartState> {
  final GetCartUseCase _getCartUseCase;
  final AddToCartUseCase _addToCartUseCase;
  final DeleteCartItemUseCase _deleteCartItemUseCase;

  CartCubit(
    this._getCartUseCase,
    this._addToCartUseCase,
    this._deleteCartItemUseCase,
  ) : super(const CartState());

  Future<void> getCart() async {
    emit(state.copyWith(status: CartStatus.loading, clearError: true));

    final result = await _getCartUseCase();

    if (isClosed) return;

    if (result is Success<List<CartItemEntity>>) {
      emit(
        state.copyWith(
          status: CartStatus.success,
          items: result.data,
          clearError: true,
        ),
      );
    } else if (result is Error<List<CartItemEntity>>) {
      emit(
        state.copyWith(
          status: CartStatus.failure,
          errorMessage: result.messageError,
        ),
      );
    }
  }

  Future<ResultApi<void>> addToCart(int productId) async {
    emit(state.copyWith(isAdding: true, clearError: true));

    try {
      final result = await _addToCartUseCase(productId: productId);

      if (isClosed) return result;

      if (result is Success<void>) {
        emit(state.copyWith(isAdding: false));

        await getCart();
      } else if (result is Error<void>) {
        emit(
          state.copyWith(isAdding: false, errorMessage: result.messageError),
        );
      }

      return result;
    } catch (e) {
      if (!isClosed) {
        emit(state.copyWith(isAdding: false, errorMessage: e.toString()));
      }

      return Error<void>(e.toString());
    }
  }

  Future<ResultApi<void>> deleteCartItem(int productId) async {
    emit(state.copyWith(deletingProductId: productId, clearError: true));

    try {
      final result = await _deleteCartItemUseCase(productId: productId);

      if (isClosed) return result;

      if (result is Success<void>) {
        final updatedItems = state.items
            .where((item) => item.productId != productId)
            .toList();

        emit(
          state.copyWith(
            status: CartStatus.success,
            items: updatedItems,
            clearDeletingProductId: true,
            clearError: true,
          ),
        );
      } else if (result is Error<void>) {
        emit(
          state.copyWith(
            clearDeletingProductId: true,
            errorMessage: result.messageError,
          ),
        );
      }

      return result;
    } catch (e) {
      if (!isClosed) {
        emit(
          state.copyWith(
            clearDeletingProductId: true,
            errorMessage: e.toString(),
          ),
        );
      }

      return Error<void>(e.toString());
    }
  }

  void increaseQuantity(int productId) {
    final updatedItems = state.items.map((item) {
      if (item.productId == productId) {
        return item.copyWith(quantity: item.quantity + 1);
      }

      return item;
    }).toList();

    emit(state.copyWith(items: updatedItems));
  }

  void decreaseQuantity(int productId) {
    final updatedItems = state.items.map((item) {
      if (item.productId == productId && item.quantity > 1) {
        return item.copyWith(quantity: item.quantity - 1);
      }

      return item;
    }).toList();

    emit(state.copyWith(items: updatedItems));
  }
}

import 'package:shopping_app/features/cart/domain/entities/cart_item_entity.dart';

enum CartStatus { initial, loading, success, failure }

class CartState {
  final CartStatus status;
  final List<CartItemEntity> items;
  final String? errorMessage;
  final bool isAdding;
  final int? deletingProductId;

  const CartState({
    this.status = CartStatus.initial,
    this.items = const [],
    this.errorMessage,
    this.isAdding = false,
    this.deletingProductId,
  });

  double get subtotal {
    return items.fold<double>(0, (sum, item) => sum + item.totalPrice);
  }

  int get totalQuantity {
    return items.fold<int>(0, (sum, item) => sum + item.quantity);
  }

  CartState copyWith({
    CartStatus? status,
    List<CartItemEntity>? items,
    String? errorMessage,
    bool clearError = false,
    bool? isAdding,
    int? deletingProductId,
    bool clearDeletingProductId = false,
  }) {
    return CartState(
      status: status ?? this.status,
      items: items ?? this.items,
      errorMessage: clearError ? null : errorMessage ?? this.errorMessage,
      isAdding: isAdding ?? this.isAdding,
      deletingProductId: clearDeletingProductId
          ? null
          : deletingProductId ?? this.deletingProductId,
    );
  }
}

class ProductResponseDto {
  List<ProductItemDto>? list;
  int? total;
  int? skip;
  int? limit;

  ProductResponseDto({this.list, this.total, this.skip, this.limit});

  ProductResponseDto.fromJson(Map<String, dynamic> json) {
    if (json['list'] != null) {
      list = <ProductItemDto>[];

      json['list'].forEach((v) {
        list!.add(ProductItemDto.fromJson(v));
      });
    }

    total = json['total'];
    skip = json['skip'];
    limit = json['limit'];
  }

  Map<String, dynamic> toJson() {
    return {
      'list': list?.map((v) => v.toJson()).toList(),
      'total': total,
      'skip': skip,
      'limit': limit,
    };
  }
}

String getFriendlyErrorMessage(Object error) {
  if (error is DioException) {
    if (error.type == DioExceptionType.connectionError ||
        error.type == DioExceptionType.connectionTimeout ||
        error.type == DioExceptionType.receiveTimeout ||
        error.type == DioExceptionType.sendTimeout) {
      return 'Check your internet connection and try again.';
    }

    if (error.type == DioExceptionType.badResponse) {
      final statusCode = error.response?.statusCode;

      if (statusCode == 401 || statusCode == 403) {
        return 'Please log in to continue.';
      }

      if (statusCode != null && statusCode >= 500) {
        return 'The server is temporarily unavailable. Try again later.';
      }
    }

    if (error.type == DioExceptionType.cancel) {
      return 'The request was cancelled.';
    }

    if (error.type == DioExceptionType.badCertificate) {
      return 'A secure connection could not be established.';
    }
  }

  return 'Something went wrong. Please try again.';
}

@injectable
class HomeCubit extends Cubit<HomeState> {
  final GetAllProductUseCase _getAllProductUseCase;
  final GetAllCategoriesUseCase _getAllCategoriesUseCase;

  static const int _pageSize = 10;

  HomeCubit(this._getAllProductUseCase, this._getAllCategoriesUseCase)
    : super(const HomeState());

  Future<void> intent(HomeIntent intent) async {
    if (intent is HProducts) {
      await _getProduct();
    } else if (intent is HCategories) {
      await _getCategories();
    }
  }

  Future<void> loadHome() async {
    await Future.wait([_getProduct(), _getCategories()]);
  }

  Future<void> _getProduct() async {
    if (state.isProductsLoading) return;

    emit(state.copyWith(isProductsLoading: true, clearProductError: true));

    try {
      final result = await _getAllProductUseCase.invoke(
        skip: 0,
        limit: _pageSize,
      );

      if (isClosed) return;

      switch (result) {
        case Success<ProductResponseEntity>():
          final response = result.data;
          final products = response.list;

          final hasMore = response.total > 0
              ? products.length < response.total
              : products.length >= _pageSize;

          emit(
            state.copyWith(
              isProductsLoading: false,
              products: products,
              hasMoreProducts: hasMore,
              clearProductError: true,
            ),
          );

        case Error<ProductResponseEntity>():
          emit(
            state.copyWith(
              isProductsLoading: false,
              productError: result.messageError,
            ),
          );
      }
    } catch (e) {
      if (isClosed) return;

      emit(
        state.copyWith(isProductsLoading: false, productError: e.toString()),
      );
    }
  }

  Future<void> loadMoreProducts() async {
    if (state.isProductsLoading ||
        state.isLoadingMore ||
        !state.hasMoreProducts ||
        state.products.isEmpty) {
      return;
    }

    emit(state.copyWith(isLoadingMore: true));

    try {
      final result = await _getAllProductUseCase.invoke(
        skip: state.products.length,
        limit: _pageSize,
      );

      if (isClosed) return;

      switch (result) {
        case Success<ProductResponseEntity>():
          final response = result.data;
          final existingIds = state.products.map((p) => p.id).toSet();

          final newProducts = response.list
              .where((product) => !existingIds.contains(product.id))
              .toList();

          final allProducts = [...state.products, ...newProducts];

          final hasMore = response.total > 0
              ? allProducts.length < response.total
              : response.list.length >= _pageSize;

          emit(
            state.copyWith(
              isLoadingMore: false,
              products: allProducts,
              hasMoreProducts: hasMore,
              clearProductError: true,
            ),
          );

        case Error<ProductResponseEntity>():
          emit(
            state.copyWith(
              isLoadingMore: false,
              productError: result.messageError,
            ),
          );
      }
    } catch (e) {
      if (isClosed) return;

      emit(state.copyWith(isLoadingMore: false, productError: e.toString()));
    }
  }

  Future<void> _getCategories() async {
    if (state.isCategoriesLoading) return;

    emit(state.copyWith(isCategoriesLoading: true, clearCategoriesError: true));

    try {
      final result = await _getAllCategoriesUseCase.invoke();

      if (isClosed) return;

      switch (result) {
        case Success<List<CategoryEntity>>():
          emit(
            state.copyWith(
              isCategoriesLoading: false,
              categories: result.data,
              clearCategoriesError: true,
            ),
          );

        case Error<List<CategoryEntity>>():
          emit(
            state.copyWith(
              isCategoriesLoading: false,
              categoriesError: result.messageError,
            ),
          );
      }
    } catch (e) {
      if (isClosed) return;

      emit(
        state.copyWith(
          isCategoriesLoading: false,
          categoriesError: e.toString(),
        ),
      );
    }
  }
}

import 'package:flutter_bloc/flutter_bloc.dart';
import 'package:injectable/injectable.dart';
import 'package:shopping_app/features/products_by_category/domain/use_case/get_products_by_category_use_case.dart';

@injectable
class ProductsByCategoryCubit extends Cubit<ProductsByCategoryState> {
  final GetProductsByCategoryUseCase _useCase;

  static const int _pageSize = 10;

  String? _currentSlug;

  ProductsByCategoryCubit(this._useCase) : super(ProductsByCategoryInitial());

  Future<void> getProductsByCategory(String slug) async {
    if (isClosed) return;

    _currentSlug = slug;

    emit(ProductsByCategoryLoading());

    try {
      final page = await _useCase.call(slug, skip: 0, limit: _pageSize);

      emit(
        ProductsByCategorySuccess(
          page.products,
          hasMoreProducts:
              page.products.isNotEmpty && page.products.length < page.total,
        ),
      );
    } catch (e) {
      emit(ProductsByCategoryError(e.toString()));
    }
  }

  Future<void> loadMoreProducts() async {
    final currentState = state;

    if (currentState is! ProductsByCategorySuccess ||
        currentState.isLoadingMore ||
        !currentState.hasMoreProducts ||
        _currentSlug == null) {
      return;
    }

    final currentProducts = currentState.products;

    emit(currentState.copyWith(isLoadingMore: true, clearLoadMoreError: true));

    try {
      final page = await _useCase.call(
        _currentSlug!,
        skip: currentProducts.length,
        limit: _pageSize,
      );

      final existingIds = currentProducts.map((p) => p.id).toSet();
      final newProducts = page.products
          .where((product) => existingIds.add(product.id))
          .toList();

      final updatedProducts = [...currentProducts, ...newProducts];

      emit(
        ProductsByCategorySuccess(
          updatedProducts,
          hasMoreProducts:
              updatedProducts.length < page.total && page.products.isNotEmpty,
        ),
      );
    } catch (e) {
      emit(
        currentState.copyWith(
          isLoadingMore: false,
          loadMoreError: e.toString(),
        ),
      );
    }
  }

  Future<void> retryLoadMore() => loadMoreProducts();
}

import 'dart:async';

import 'package:flutter/foundation.dart';
import 'package:flutter_bloc/flutter_bloc.dart';
import 'package:injectable/injectable.dart';

import '../../domain/entities/search_page_entity.dart';
import '../../domain/use_case/search_products_use_case.dart';
import 'search_state.dart';

@injectable
class SearchCubit extends Cubit<SearchState> {
  final SearchProductsUseCase _searchProductsUseCase;

  Timer? _debounce;
  int _requestId = 0;

  SearchCubit(this._searchProductsUseCase) : super(SearchInitial());

  void searchProducts(String query) {
    _debounce?.cancel();

    final currentRequestId = ++_requestId;
    final searchQuery = query.trim();

    if (searchQuery.isEmpty) {
      emit(SearchInitial());
      return;
    }

    _debounce = Timer(const Duration(milliseconds: 500), () {
      _performSearch(searchQuery, currentRequestId);
    });
  }

  Future<void> _performSearch(String query, int requestId) async {
    emit(SearchLoading());

    try {
      debugPrint('Searching for: $query');

      final result = await _searchProductsUseCase.call(query);

      // Ignore results from older requests.
      if (isClosed || requestId != _requestId) return;

      final normalizedQuery = query.toLowerCase();

      final filteredProducts = result.products.where((product) {
        return product.title.toLowerCase().contains(normalizedQuery) ||
            product.description.toLowerCase().contains(normalizedQuery);
      }).toList();

      debugPrint('Products from API: ${result.products.length}');
      debugPrint('Products after filtering: ${filteredProducts.length}');

      if (filteredProducts.isEmpty) {
        emit(SearchEmpty());
      } else {
        emit(
          SearchSuccess(
            SearchPageEntity(
              products: filteredProducts,
              total: filteredProducts.length,
              skip: result.skip,
              limit: result.limit,
            ),
          ),
        );
      }
    } catch (e, stackTrace) {
      if (isClosed || requestId != _requestId) return;

      debugPrint('Search error: $e');
      debugPrint('$stackTrace');

      emit(SearchError('Something went wrong. Please try again.'));
    }
  }

  @override
  Future<void> close() {
    _debounce?.cancel();
    return super.close();
  }
}
